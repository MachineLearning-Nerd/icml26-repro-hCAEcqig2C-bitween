\[
\tan\bigl(f(x)\bigr)=\sinh(x).

\cosh x = \sqrt{1+\sinh^{2}x}= \sqrt{1+\tan^{2}f(x)} = \sec\bigl(f(x)\bigr),
\]

\[

\tan\bigl(f(x+r)\bigr) &= \tan\bigl(f(x)\bigr)\cosh r+\sec\bigl(f(x)\bigr)\sinh r,\\[2mm]
\tan\bigl(f(x-r)\bigr) &= \tan\bigl(f(x)\bigr)\cosh r-\sec\bigl(f(x)\bigr)\sinh r.
\end{aligned}
\]

\[

\sin\bigl(f(x-r)\bigr)=\sin\bigl(f(x)\bigr)\cosh r-\sinh r .
